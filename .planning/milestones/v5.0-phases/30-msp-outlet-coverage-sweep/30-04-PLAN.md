```python
# scripts/audit_signal_role.py — NEW function (Plan 30-04)
def _propose_inherited_roles(
    db: ObjectDatabase,
    mc_name: str,
) -> list[dict] | None:
    """Propose signal_role for an MC/MCS variant by mirroring its bare-MSP
    sibling outlet-by-outlet (positional).

    Returns:
      None if no sibling exists or I/O parity fails (caller falls through
      to the digest classifier).
      Else: list of {outlet_id, signal_role, confidence, source} dicts,
      one per outlet of the MC variant. Trailing outlets where the MC has
      MORE outlets than the sibling are emitted with `signal_role: None`
      and `confidence: "trailing-fallthrough"` so the caller can route
      them to the digest classifier.

    Parity gate (per threat-mitigation): the sibling's INLET count AND
    OUTLET count must equal the MC variant's, otherwise the I/O layout is
    divergent enough that positional copying is unsafe — return None and
    let the digest classifier propose roles for the entire object.

    Naming convention (per CONTEXT D-11): strip `mc.` or `mcs.` prefix,
    then look up via db.lookup(bare_name). The lookup resolves aliases,
    so an alias-mediated match is also valid (the sibling is whatever
    db.lookup returns — never raw JSON keys).

    Confidence tiers emitted by this function:
      - "inherited"            -> sibling outlet had a signal_role and we copied it
      - "inherited-no-role"    -> sibling outlet exists but has no signal_role yet
                                   (caller falls through to digest classifier)
      - "trailing-fallthrough" -> MC has more outlets than sibling at this index
    """
```

```python
def _propose_inherited_roles(
    db: "ObjectDatabase",
    mc_name: str,
) -> list[dict] | None:
    """Propose signal_role for an MC variant by mirroring its bare-MSP
    sibling outlet-by-outlet (positional). See module docstring §D-11.

    Returns None when:
      - mc_name does not start with `mc.` or `mcs.`
      - the stripped name does not resolve via db.lookup (no sibling)
      - sibling and MC variant have different inlet OR outlet counts
        (parity gate — divergence is unsafe to mirror positionally)

    Otherwise returns a list of dicts (one per MC outlet):
      {"outlet_id": int, "signal_role": str | None, "confidence": str,
       "source": "sibling-mirror"}

    confidence values:
      - "inherited"           : sibling outlet had a signal_role; copied verbatim
      - "inherited-no-role"   : sibling outlet had no signal_role; signal_role=None
      - "trailing-fallthrough": MC has MORE outlets than sibling at this index
                                 (only emitted when MC has more outlets than sibling
                                 BUT all earlier outlets parity-match -- a soft form
                                 of parity that the MAIN parity gate rejects today;
                                 reserved for a future relaxation. With the strict
                                 parity gate above, this branch is currently
                                 unreachable but the test fixture covers it for
                                 forward compatibility.)
    """
    # Strip prefix (mcs first - longer prefix wins)
    if mc_name.startswith("mcs."):
        bare_name = mc_name[len("mcs."):]
    elif mc_name.startswith("mc."):
        bare_name = mc_name[len("mc."):]
    else:
        return None

    sibling = db.lookup(bare_name)
    if sibling is None:
        return None

    mc_obj = db.lookup(mc_name)
    if mc_obj is None:
        return None

    # Parity gate: inlet count AND outlet count must match.
    # Per threat model T-30-04-01: name-only matches with mismatched I/O
    # are coincidental — fall through to the classifier rather than copying
    # potentially-wrong roles.
    sibling_inlets = sibling.get("inlets", [])
    sibling_outlets = sibling.get("outlets", [])
    mc_inlets = mc_obj.get("inlets", [])
    mc_outlets = mc_obj.get("outlets", [])
    if len(sibling_inlets) != len(mc_inlets):
        return None
    if len(sibling_outlets) != len(mc_outlets):
        return None

    rows: list[dict] = []
    for i, mc_outlet in enumerate(mc_outlets):
        sib_role = sibling_outlets[i].get("signal_role") if i < len(sibling_outlets) else None
        if sib_role is None:
            rows.append({
                "outlet_id": mc_outlet.get("id", i),
                "signal_role": None,
                "confidence": "inherited-no-role",
                "source": "sibling-mirror",
            })
        else:
            rows.append({
                "outlet_id": mc_outlet.get("id", i),
                "signal_role": sib_role,
                "confidence": "inherited",
                "source": "sibling-mirror",
            })

    return rows
```
