        if isinstance(original, dict) and isinstance(result, dict):
            for key in set(list(original.keys()) + list(result.keys())):
                if key not in original:
                    print(f"{path}.{key}: ADDED (value={result[key]!r})")
                elif key not in result:
                    print(f"{path}.{key}: MISSING")
                else:
                    _find_diffs(original[key], result[key], f"{path}.{key}")
        elif isinstance(original, list) and isinstance(result, list):
            if len(original) != len(result):

            for i, (a, b) in enumerate(zip(original, result)):
                _find_diffs(a, b, f"{path}[{i}]")
        elif original != result:
            print(f"{path}: {original!r} != {result!r}")
