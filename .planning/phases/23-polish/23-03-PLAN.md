    def _check_missing_info_text(boxes: list[dict]) -> list[CriticResult]:
      - Iterate boxes (non-recursive -- review_m4l only passes top-level boxes)
      - For each box with maxclass starting with "live." and not in _LIVE_NO_PARAM:
        - If parameter_enable is set and box.get("annotation") is falsy:
          - Append CriticResult("warning", "{maxclass} ({box_id}) missing info text -- Ableton Info View will show no description", "Run polish_m4l_device() or set annotation attribute manually")
      - Recurse into subpatcher boxes
      - Return results

    def _check_missing_live_banks(boxes: list[dict]) -> list[CriticResult]:
      - Check if any live.* parameter controls exist (maxclass starts with "live." and not in _LIVE_NO_PARAM and has parameter_enable)
      - Check if live.banks box exists (maxclass == "live.banks")
      - If parameter controls exist but no live.banks:
        - Return [CriticResult("warning", "M4L device has parameter controls but no live.banks -- Push controller layout will use Ableton defaults", "Run polish_m4l_device() or add live.banks for explicit Push bank control")]
      - Else return []

    Wire both into review_m4l():
      results.extend(_check_missing_info_text(boxes))
      results.extend(_check_missing_live_banks(boxes))
