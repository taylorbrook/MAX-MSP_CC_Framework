import json, glob, sys
files = glob.glob('patterns/gen/**/*.gendsp', recursive=True)
if len(files) < 18:
    print(f'FAIL: Only {len(files)} .gendsp files, expected 18+')
    sys.exit(1)
errors = []
for f in sorted(files):
    try:
        d = json.load(open(f))
        p = d['patcher']
        boxes = [b['box'] for b in p['boxes']]
        has_codebox = any(b.get('maxclass') == 'codebox' for b in boxes)
        has_in = any('in ' in b.get('text', '') for b in boxes if b.get('maxclass') == 'newobj')
        has_out = any('out ' in b.get('text', '') for b in boxes if b.get('maxclass') == 'newobj')
        has_lines = len(p.get('lines', [])) > 0
        if not (has_codebox and has_in and has_out and has_lines):
            errors.append(f'{f}: missing codebox/in/out/lines')
    except Exception as e:
        errors.append(f'{f}: {e}')
if errors:
    print('ERRORS:'); [print(f'  {e}') for e in errors]; sys.exit(1)
print(f'OK: {len(files)} valid .gendsp files')
import os
assert os.path.exists('patterns/gen/INDEX.md'), 'INDEX.md missing'
print('OK: INDEX.md exists')
